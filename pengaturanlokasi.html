Pengaturan Lokasi Simpan

Tentukan folder Google Drive tujuan penyimpanan berkas. Berkas yang diunggah akan masuk ke folder ini lewat Google Apps Script.

Tujuan penyimpanan

Belum disimpan
Buka folder
Cara menyiapkan Google Drive dan script
  1. Buat dan bagikan folder

    Di Google Drive, buat folder tujuan. Klik kanan folder > Bagikan > Akses umum > pilih Siapa saja yang memiliki link sebagai Pelihat. Salin link folder dan tempel di kolom pertama.

  2. Pasang kode script

    Buka script.google.com, buat proyek baru, hapus isi bawaan, lalu tempel kode dari bagian “Kode Google Apps Script” di bawah.

  3. Terapkan sebagai aplikasi web

    Klik Terapkan > Penerapan baru > Aplikasi web. Atur Jalankan sebagai: Saya dan Yang memiliki akses: Siapa saja. Izinkan akses Google Drive, lalu salin URL yang berakhiran /exec.

  4. Tempel URL dan uji

    Tempel URL di kolom kedua, klik Uji koneksi. Jika nama folder muncul, klik Simpan pengaturan.

Setelah mengubah kode script, buat Penerapan baru agar perubahan berlaku di URL /exec. Folder harus milik akun yang menerapkan script, atau dibagikan ke akun itu sebagai Editor.
Kode Google Apps Script
Tempel di editor script (Code.gs)
/**
 * Upload Berkas ke Google Drive (PKKS)
 *
 * CARA PASANG
 * 1. Buka https://script.google.com > Proyek baru, hapus isi bawaan, tempel seluruh kode ini.
 * 2. Klik Terapkan (Deploy) > Penerapan baru (New deployment) > jenis "Aplikasi web" (Web app).
 *      - Jalankan sebagai (Execute as)  : Saya (Me)
 *      - Yang memiliki akses (Access)   : Siapa saja (Anyone)
 * 3. Klik Terapkan, izinkan akses Google Drive, lalu salin "URL aplikasi web" (berakhiran /exec).
 * 4. Tempel URL itu di halaman Pengaturan Lokasi Simpan, bersama link folder Google Drive.
 *
 * Setiap kali kode ini diubah, buat Penerapan baru (atau Kelola penerapan > Edit > Versi baru)
 * agar perubahan berlaku di URL /exec.
 */

// Kunci rahasia (opsional). Jika diisi, halaman harus mengirim kunci yang sama.
const TOKEN = '';

// ID folder bawaan (opsional). Dipakai bila halaman tidak mengirim folderId.
const FOLDER_ID_DEFAULT = '';

// true = file yang diunggah dibuat "Siapa saja yang memiliki link" (hanya melihat)
const SHARE_ANYONE = true;

function out_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function auth_(token) {
  if (TOKEN && token !== TOKEN) throw new Error('Kunci rahasia tidak cocok.');
}

function folder_(id, sub) {
  id = id || FOLDER_ID_DEFAULT;
  if (!id) throw new Error('ID folder kosong.');
  var folder = DriveApp.getFolderById(id);
  if (sub) {
    sub = String(sub).replace(/[\\\/:*?"<>|]/g, '-').trim().slice(0, 80);
    if (sub) {
      var it = folder.getFoldersByName(sub);
      folder = it.hasNext() ? it.next() : folder.createFolder(sub);
    }
  }
  return folder;
}

// Uji koneksi: ?folderId=...&token=...
function doGet(e) {
  try {
    var p = (e && e.parameter) || {};
    auth_(p.token);
    var f = folder_(p.folderId);
    return out_({ ok: true, folderId: f.getId(), folderName: f.getName(), url: f.getUrl() });
  } catch (err) {
    return out_({ ok: false, error: String(err && err.message || err) });
  }
}

// Unggah file. Body JSON (dikirim sebagai text/plain):
// { folderId, token, name, mimeType, data (base64), subfolder (opsional) }
function doPost(e) {
  try {
    var d = JSON.parse(e.postData.contents);
    auth_(d.token);

    if (!d.name || !d.data) throw new Error('Nama atau isi file kosong.');

    var bytes = Utilities.base64Decode(d.data);
    var blob = Utilities.newBlob(bytes, d.mimeType || 'application/octet-stream', d.name);
    var file = folder_(d.folderId, d.subfolder).createFile(blob);

    if (SHARE_ANYONE) {
      try {
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      } catch (x) { /* dibatasi kebijakan domain: abaikan */ }
    }

    return out_({
      ok: true,
      id: file.getId(),
      name: file.getName(),
      size: file.getSize(),
      url: file.getUrl(),
      downloadUrl: 'https://drive.google.com/uc?export=download&id=' + file.getId()
    });
  } catch (err) {
    return out_({ ok: false, error: String(err && err.message || err) });
  }
}
